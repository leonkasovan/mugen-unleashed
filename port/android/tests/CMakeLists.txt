cmake_minimum_required(VERSION 3.22.1)
project(android_native_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
find_path(SDL2_MIXER_INCLUDE_DIR SDL_mixer.h PATH_SUFFIXES SDL2 REQUIRED)
find_path(SDL2_TTF_INCLUDE_DIR SDL_ttf.h PATH_SUFFIXES SDL2 REQUIRED)
find_library(SDL2_LIBRARY NAMES SDL2 SDL2-2.0 REQUIRED)
find_library(SDL2_MIXER_LIBRARY NAMES SDL2_mixer REQUIRED)
find_library(SDL2_TTF_LIBRARY NAMES SDL2_ttf REQUIRED)
file(GLOB ENGINE "${ROOT}/src/*.c")
list(FILTER ENGINE EXCLUDE REGEX ".*/(sdlmain|glrender|gldetect|sdl2compat|sdlmixercompat)\\.c$")
file(GLOB LUA "${ROOT}/third_party/lua-5.1.4/src/*.c")
list(FILTER LUA EXCLUDE REGEX ".*/(lua|luac)\\.c$")
file(GLOB PNG "${ROOT}/third_party/libpng-1.2.38/png*.c")
list(FILTER PNG EXCLUDE REGEX ".*/(pnggccrd|pngvcrd)\\.c$")
file(GLOB RESAMPLE "${ROOT}/third_party/libresample/src/*.c")
file(GLOB ZIP "${ROOT}/third_party/minizip-0.15/*.c")
file(GLOB PLATFORM "${ROOT}/port/android/app/src/main/jni/src/android_*.c")
list(FILTER PLATFORM EXCLUDE REGEX ".*/android_main\\.c$")
add_library(android_native SHARED ${ENGINE} ${LUA} ${PNG} ${RESAMPLE} ${ZIP} ${PLATFORM}
 "${CMAKE_CURRENT_LIST_DIR}/test_present_host.c")
target_compile_definitions(android_native PRIVATE ANDROID_PORT=1 MUGEN_ANDROID_SW=1 LIBRESAMPLE_STATIC=1)
target_include_directories(android_native PRIVATE
 "${ROOT}/src" "${ROOT}/port/android/app/src/main/jni/src"
 "${ROOT}/third_party/libpng-1.2.38" "${ROOT}/third_party/lua-5.1.4/src"
 "${ROOT}/third_party/minizip-0.15" "${ROOT}/third_party/libresample/include" "${ROOT}/third_party/libresample/src"
 "${SDL2_INCLUDE_DIR}" "${SDL2_MIXER_INCLUDE_DIR}" "${SDL2_TTF_INCLUDE_DIR}")
target_compile_options(android_native PRIVATE -O1 -g -Werror=implicit-function-declaration -fno-strict-aliasing)
target_link_options(android_native PRIVATE -Wl,--no-undefined -Wl,--wrap=SDL_UpdateTexture -Wl,--wrap=SDL_RenderPresent)
target_link_libraries(android_native PRIVATE
 "${SDL2_LIBRARY}" "${SDL2_MIXER_LIBRARY}" "${SDL2_TTF_LIBRARY}" z m dl pthread)
add_executable(test_native_sdl2 "${CMAKE_CURRENT_LIST_DIR}/test_native_sdl2.c")
target_compile_definitions(test_native_sdl2 PRIVATE ANDROID_PORT=1 SDL_MAIN_HANDLED)
target_compile_options(test_native_sdl2 PRIVATE -UNDEBUG)
target_include_directories(test_native_sdl2 PRIVATE
 "${ROOT}/src" "${ROOT}/port/android/app/src/main/jni/src"
 "${ROOT}/third_party/libpng-1.2.38" "${ROOT}/third_party/minizip-0.15"
 "${SDL2_INCLUDE_DIR}" "${SDL2_MIXER_INCLUDE_DIR}" "${SDL2_TTF_INCLUDE_DIR}")
target_link_libraries(test_native_sdl2 PRIVATE android_native "${SDL2_LIBRARY}" "${SDL2_MIXER_LIBRARY}" "${SDL2_TTF_LIBRARY}")
option(NATIVE_SDL2_SANITIZE "Check native engine tests with ASan/UBSan" OFF)
if(NATIVE_SDL2_SANITIZE)
 foreach(target android_native test_native_sdl2)
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
enable_testing()
find_file(NATIVE_SDL2_TEST_FONT DejaVuSans.ttf PATHS /usr/share/fonts/truetype/dejavu)
if(NATIVE_SDL2_TEST_FONT)
 add_test(NAME native_sdl2 COMMAND test_native_sdl2 "${NATIVE_SDL2_TEST_FONT}")
else()
 add_test(NAME native_sdl2 COMMAND test_native_sdl2)
endif()
set_tests_properties(native_sdl2 PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
